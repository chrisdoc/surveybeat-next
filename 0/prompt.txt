GERMAN_PHONE_PATTERN =
  /(?&lt;![\w+])(?:(?:\+|00)49[ /-]?(?:\(0\) ?)?|\(?0)(?:1[5-7]\d|[2-9]\d{1,4})\)?(?:(?: [-/] |[ /-])?\d{1,6}){1,6}(?!\w|[.,]\d)/g;

DATE_OR_YEAR_RANGE_PATTERNS = [
  /^\d{1,2}\.\d{1,2}\.(?:\d{4}|\d{2})(?!\d)/,
  /^\d{4}-\d{1,2}-\d{1,2}(?!\d)/,
  /^\d{1,2}\/\d{1,2}\/\d{2,4}(?!\d)/,
  /^\d{4} ?[-\u2013] ?\d{4}$/,
];
